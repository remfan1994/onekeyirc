Execute apply.sh numa fonte inglesa r1 limpa antes da compilação; não sobreponha pacotes.

  tar -xzf onekeyirc-1.0-a.tar.gz
  tar -xzf onekeyirc-languages-1.0-a.tar.gz
  ONEKEYIRC_LANG=pt sh onekeyirc-languages-1.0-a/apply.sh onekeyirc-1.0-a
  cd onekeyirc-1.0-a
  ONEKEYIRC_LANG=pt ./configure && make
  ONEKEYIRC_LANG=pt ./onekeyirc --version
